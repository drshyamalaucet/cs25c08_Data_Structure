For an absolute path, the canonical result always begins with /.

6. Pseudocode

``` 
function simplifyPath(path):

    stack = empty stack

    components = split path using "/"

    for each component in components:

        if component == "" or component == ".":
            continue

        else if component == "..":

            if stack is not empty:
                pop stack

        else:
            push component onto stack

    if stack is empty:
        return "/"

    return "/" + join(stack, "/")
```

7. Python Implementation

``` 
def simplify_path(path):
    stack = []

    for part in path.split("/"):

        if part == "" or part == ".":
            continue

        if part == "..":

            if stack:
                stack.pop()

        else:
            stack.append(part)

    return "/" + "/".join(stack)


# Example
path = "/home//foo/../bar/"

print(simplify_path(path))

# Output:
# /home/bar
```

8. C++ Implementation

``` 
#include <iostream>
#include <sstream>
#include <stack>
#include <string>

using namespace std;

string simplifyPath(string path) {

    stack<string> st;

    string component;

    stringstream ss(path);

    while (getline(ss, component, '/')) {

        if (component == "" || component == ".") {
            continue;
        }

        if (component == "..") {

            if (!st.empty()) {
                st.pop();
            }
        }

        else {
            st.push(component);
        }
    }

    string result = "";

    while (!st.empty()) {

        result = "/" + st.top() + result;
        st.pop();
    }

    if (result.empty()) {
        return "/";
    }

    return result;
}

int main() {

    string path = "/home//foo/../bar/";

    cout << simplifyPath(path);

    return 0;
}

// Output:
// /home/bar
```

10. Complexity Analysis

  • Time Complexity: O(n)
  • Space Complexity: O(n)

Each path component is processed once. The stack stores directory names that remain in the canonical path.
