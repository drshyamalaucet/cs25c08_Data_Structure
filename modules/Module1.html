1. Linear Lists ADT

A Linear List is an ordered collection of elements in which each element has a unique predecessor and successor, except the first and last elements.

Array-based List

Elements are stored in contiguous memory locations.

Singly Linked List

Each node stores data and a pointer to the next node.

Doubly Linked List

Each node contains previous and next pointers.

Circular Linked List

The last node points back to the first node.

ADT: Abstract Data Type defines what operations can be performed on a data structure without specifying how those operations are implemented internally.

Common ADT Operations

Operation Description
Insert BeginAdd an element at the beginning.
Insert MiddleAdd an element at a specified position.
Insert EndAdd an element at the end.
Delete BeginRemove the first element.
Delete MiddleRemove an element from a specified position.
Delete EndRemove the last element.
SearchFind an element.
TraversalVisit all elements.

2. Array-Based Linear List ADT

An array-based list stores elements in consecutive memory locations. Insertion and deletion may require shifting elements.

Array List Pseudocode

INSERT_END(A, n, value)
    A[n] ← value
    n ← n + 1

INSERT_BEGIN(A, n, value)
    FOR i ← n DOWNTO 1
        A[i] ← A[i-1]
    A[0] ← value
    n ← n + 1

INSERT_MIDDLE(A, n, pos, value)
    FOR i ← n DOWNTO pos
        A[i] ← A[i-1]
    A[pos] ← value
    n ← n + 1

DELETE_BEGIN(A, n)
    FOR i ← 0 TO n-2
        A[i] ← A[i+1]
    n ← n - 1

DELETE_END(A, n)
    n ← n - 1

DELETE_MIDDLE(A, n, pos)
    FOR i ← pos TO n-2
        A[i] ← A[i+1]
    n ← n - 1

SEARCH(A, n, key)
    FOR i ← 0 TO n-1
        IF A[i] = key
            RETURN i
    RETURN -1

TRAVERSAL(A, n)
    FOR i ← 0 TO n-1
        PRINT A[i]

C++ Implementation

#include <iostream>
using namespace std;

class ArrayList {
    int a[100];
    int n;

public:
    ArrayList() {
        n = 0;
    }

    void insertBegin(int value) {
        for (int i = n; i > 0; i--)
            a[i] = a[i - 1];

        a[0] = value;
        n++;
    }

    void insertEnd(int value) {
        a[n++] = value;
    }

    void insertMiddle(int pos, int value) {
        if (pos < 0 || pos > n) return;

        for (int i = n; i > pos; i--)
            a[i] = a[i - 1];

        a[pos] = value;
        n++;
    }

    void deleteBegin() {
        if (n == 0) return;

        for (int i = 0; i < n - 1; i++)
            a[i] = a[i + 1];

        n--;
    }

    void deleteEnd() {
        if (n > 0)
            n--;
    }

    void deleteMiddle(int pos) {
        if (pos < 0 || pos >= n) return;

        for (int i = pos; i < n - 1; i++)
            a[i] = a[i + 1];

        n--;
    }

    int search(int key) {
        for (int i = 0; i < n; i++)
            if (a[i] == key)
                return i;

        return -1;
    }

    void traverse() {
        for (int i = 0; i < n; i++)
            cout << a[i] << " ";
        cout << endl;
    }
};

3. Singly Linked List

A singly linked list consists of nodes. Each node contains:

Data

The actual value stored in the node.

Next

Address of the next node.

Pseudocode – Singly Linked List

INSERT_BEGIN(value)
    newNode ← CREATE NODE(value)
    newNode.next ← head
    head ← newNode

INSERT_END(value)
    newNode ← CREATE NODE(value)

    IF head = NULL
        head ← newNode
        RETURN

    temp ← head
    WHILE temp.next ≠ NULL
        temp ← temp.next

    temp.next ← newNode

INSERT_MIDDLE(pos, value)
    newNode ← CREATE NODE(value)

    temp ← head
    REPEAT pos-1 TIMES
        temp ← temp.next

    newNode.next ← temp.next
    temp.next ← newNode

DELETE_BEGIN()
    IF head = NULL
        RETURN

    temp ← head
    head ← head.next
    DELETE temp

DELETE_END()
    IF head = NULL
        RETURN

    IF head.next = NULL
        DELETE head
        head ← NULL
        RETURN

    temp ← head
    WHILE temp.next.next ≠ NULL
        temp ← temp.next

    DELETE temp.next
    temp.next ← NULL

SEARCH(key)
    temp ← head

    WHILE temp ≠ NULL
        IF temp.data = key
            RETURN FOUND

        temp ← temp.next

    RETURN NOT FOUND

TRAVERSAL()
    temp ← head

    WHILE temp ≠ NULL
        PRINT temp.data
        temp ← temp.next

C++ Implementation

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* next;

    Node(int value) {
        data = value;
        next = nullptr;
    }
};

class SinglyList {
    Node* head;

public:
    SinglyList() {
        head = nullptr;
    }

    void insertBegin(int value) {
        Node* newNode = new Node(value);
        newNode->next = head;
        head = newNode;
    }

    void insertEnd(int value) {
        Node* newNode = new Node(value);

        if (head == nullptr) {
            head = newNode;
            return;
        }

        Node* temp = head;

        while (temp->next != nullptr)
            temp = temp->next;

        temp->next = newNode;
    }

    void insertMiddle(int pos, int value) {
        if (pos == 0) {
            insertBegin(value);
            return;
        }

        Node* temp = head;

        for (int i = 0; i < pos - 1 && temp != nullptr; i++)
            temp = temp->next;

        if (temp == nullptr) return;

        Node* newNode = new Node(value);

        newNode->next = temp->next;
        temp->next = newNode;
    }

    void deleteBegin() {
        if (head == nullptr) return;

        Node* temp = head;
        head = head->next;
        delete temp;
    }

    void deleteEnd() {
        if (head == nullptr) return;

        if (head->next == nullptr) {
            delete head;
            head = nullptr;
            return;
        }

        Node* temp = head;

        while (temp->next->next != nullptr)
            temp = temp->next;

        delete temp->next;
        temp->next = nullptr;
    }

    void deleteMiddle(int pos) {
        if (head == nullptr) return;

        if (pos == 0) {
            deleteBegin();
            return;
        }

        Node* temp = head;

        for (int i = 0; i < pos - 1 && temp->next != nullptr; i++)
            temp = temp->next;

        if (temp->next == nullptr) return;

        Node* del = temp->next;
        temp->next = del->next;
        delete del;
    }

    bool search(int key) {
        Node* temp = head;

        while (temp != nullptr) {
            if (temp->data == key)
                return true;

            temp = temp->next;
        }

        return false;
    }

    void traverse() {
        Node* temp = head;

        while (temp != nullptr) {
            cout << temp->data << " -> ";
            temp = temp->next;
        }

        cout << "NULL";
    }
};

4. Doubly Linked List

A doubly linked list contains three parts: Previous pointer → Data → Next pointer.

Visual Structure

NULL <-- [prev | DATA | next] <-->
       [prev | DATA | next] <-->
       [prev | DATA | next] --> NULL

Pseudocode

INSERT_BEGIN(value)
    newNode ← CREATE NODE(value)
    newNode.next ← head

    IF head ≠ NULL
        head.prev ← newNode

    head ← newNode

INSERT_END(value)
    newNode ← CREATE NODE(value)

    IF head = NULL
        head ← newNode
        RETURN

    temp ← head

    WHILE temp.next ≠ NULL
        temp ← temp.next

    temp.next ← newNode
    newNode.prev ← temp

INSERT_MIDDLE(pos, value)
    Locate node at position pos-1

    newNode.next ← current.next
    newNode.prev ← current

    IF current.next ≠ NULL
        current.next.prev ← newNode

    current.next ← newNode

DELETE_BEGIN()
    temp ← head
    head ← head.next

    IF head ≠ NULL
        head.prev ← NULL

    DELETE temp

DELETE_END()
    Move to last node
    last.prev.next ← NULL
    DELETE last

SEARCH(key)
    Traverse from head
    Compare every data value
    Return FOUND or NOT FOUND

TRAVERSAL()
    Start at head
    Follow next pointers
    Print every node

C++ Implementation

#include <iostream>
using namespace std;

struct DNode {
    int data;
    DNode* prev;
    DNode* next;

    DNode(int value) {
        data = value;
        prev = nullptr;
        next = nullptr;
    }
};

class DoublyList {
    DNode* head;

public:
    DoublyList() {
        head = nullptr;
    }

    void insertBegin(int value) {
        DNode* newNode = new DNode(value);

        newNode->next = head;

        if (head != nullptr)
            head->prev = newNode;

        head = newNode;
    }

    void insertEnd(int value) {
        DNode* newNode = new DNode(value);

        if (head == nullptr) {
            head = newNode;
            return;
        }

        DNode* temp = head;

        while (temp->next != nullptr)
            temp = temp->next;

        temp->next = newNode;
        newNode->prev = temp;
    }

    void insertMiddle(int pos, int value) {
        if (pos == 0) {
            insertBegin(value);
            return;
        }

        DNode* temp = head;

        for (int i = 0; i < pos - 1 && temp != nullptr; i++)
            temp = temp->next;

        if (temp == nullptr) return;

        DNode* newNode = new DNode(value);

        newNode->next = temp->next;
        newNode->prev = temp;

        if (temp->next != nullptr)
            temp->next->prev = newNode;

        temp->next = newNode;
    }

    void deleteBegin() {
        if (head == nullptr) return;

        DNode* temp = head;
        head = head->next;

        if (head != nullptr)
            head->prev = nullptr;

        delete temp;
    }

    void deleteEnd() {
        if (head == nullptr) return;

        DNode* temp = head;

        while (temp->next != nullptr)
            temp = temp->next;

        if (temp->prev != nullptr)
            temp->prev->next = nullptr;
        else
            head = nullptr;

        delete temp;
    }

    void deleteMiddle(int pos) {
        if (head == nullptr) return;

        DNode* temp = head;

        for (int i = 0; i < pos && temp != nullptr; i++)
            temp = temp->next;

        if (temp == nullptr) return;

        if (temp->prev != nullptr)
            temp->prev->next = temp->next;
        else
            head = temp->next;

        if (temp->next != nullptr)
            temp->next->prev = temp->prev;

        delete temp;
    }

    bool search(int key) {
        DNode* temp = head;

        while (temp != nullptr) {
            if (temp->data == key)
                return true;

            temp = temp->next;
        }

        return false;
    }

    void traverse() {
        DNode* temp = head;

        while (temp != nullptr) {
            cout << temp->data << " <-> ";
            temp = temp->next;
        }

        cout << "NULL";
    }
};

5. Circular Linked List

In a circular linked list, the last node does not contain NULL. Instead, it points back to the first node.

Visual Representation

       ┌─────────────────────────────────┐
       ↓                                 │
HEAD → [10] → [20] → [30] → [40] ───────┘

Last node.next = HEAD

Pseudocode

INSERT_BEGIN(value)
    Create newNode

    IF head = NULL
        head ← newNode
        newNode.next ← head
        RETURN

    Find last node

    newNode.next ← head
    last.next ← newNode
    head ← newNode

INSERT_END(value)
    Create newNode

    IF head = NULL
        head ← newNode
        newNode.next ← head
        RETURN

    Find last node

    last.next ← newNode
    newNode.next ← head

DELETE_BEGIN()
    IF head = NULL
        RETURN

    IF head.next = head
        DELETE head
        head ← NULL
        RETURN

    Find last node
    temp ← head
    head ← head.next
    last.next ← head
    DELETE temp

SEARCH(key)
    IF head = NULL
        RETURN NOT FOUND

    temp ← head

    DO
        IF temp.data = key
            RETURN FOUND
        temp ← temp.next
    WHILE temp ≠ head

    RETURN NOT FOUND

TRAVERSAL()
    IF head = NULL
        RETURN

    temp ← head

    DO
        PRINT temp.data
        temp ← temp.next
    WHILE temp ≠ head

C++ Implementation

#include <iostream>
using namespace std;

struct CNode {
    int data;
    CNode* next;

    CNode(int value) {
        data = value;
        next = nullptr;
    }
};

class CircularList {
    CNode* head;

public:
    CircularList() {
        head = nullptr;
    }

    void insertBegin(int value) {
        CNode* newNode = new CNode(value);

        if (head == nullptr) {
            head = newNode;
            newNode->next = head;
            return;
        }

        CNode* last = head;

        while (last->next != head)
            last = last->next;

        newNode->next = head;
        last->next = newNode;
        head = newNode;
    }

    void insertEnd(int value) {
        CNode* newNode = new CNode(value);

        if (head == nullptr) {
            head = newNode;
            newNode->next = head;
            return;
        }

        CNode* last = head;

        while (last->next != head)
            last = last->next;

        last->next = newNode;
        newNode->next = head;
    }

    void insertMiddle(int pos, int value) {
        if (pos == 0) {
            insertBegin(value);
            return;
        }

        if (head == nullptr) return;

        CNode* temp = head;

        for (int i = 0; i < pos - 1; i++) {
            temp = temp->next;

            if (temp == head)
                return;
        }

        CNode* newNode = new CNode(value);

        newNode->next = temp->next;
        temp->next = newNode;
    }

    void deleteBegin() {
        if (head == nullptr) return;

        if (head->next == head) {
            delete head;
            head = nullptr;
            return;
        }

        CNode* last = head;

        while (last->next != head)
            last = last->next;

        CNode* temp = head;

        head = head->next;
        last->next = head;

        delete temp;
    }

    void deleteEnd() {
        if (head == nullptr) return;

        if (head->next == head) {
            delete head;
            head = nullptr;
            return;
        }

        CNode* temp = head;

        while (temp->next->next != head)
            temp = temp->next;

        delete temp->next;
        temp->next = head;
    }

    void deleteMiddle(int pos) {
        if (head == nullptr) return;

        if (pos == 0) {
            deleteBegin();
            return;
        }

        CNode* temp = head;

        for (int i = 0; i < pos - 1; i++) {
            temp = temp->next;

            if (temp->next == head)
                return;
        }

        CNode* del = temp->next;

        if (del == head)
            return;

        temp->next = del->next;
        delete del;
    }

    bool search(int key) {
        if (head == nullptr)
            return false;

        CNode* temp = head;

        do {
            if (temp->data == key)
                return true;

            temp = temp->next;
        } while (temp != head);

        return false;
    }

    void traverse() {
        if (head == nullptr) {
            cout << "EMPTY";
            return;
        }

        CNode* temp = head;

        do {
            cout << temp->data << " -> ";
            temp = temp->next;
        } while (temp != head);

        cout << "(HEAD)";
    }
};

6. ADT Operations – Visual Summary

Insert at Beginning

New element becomes the first element.

Before: 10 → 20 → 30

After: 5 → 10 → 20 → 30

Insert in Middle

Element is inserted at a selected position.

Before: 10 → 20 → 30

After: 10 → 15 → 20 → 30

Insert at End

Element becomes the last element.

Before: 10 → 20 → 30

After: 10 → 20 → 30 → 40

Delete Beginning

First element is removed.

Before: 10 → 20 → 30

After: 20 → 30

Delete Middle

Selected element is removed.

Before: 10 → 20 → 30

After: 10 → 30

Delete End

Last element is removed.

Before: 10 → 20 → 30

After: 10 → 20

Search

Compare the search key with every element until it is found or the list ends.

Traversal

Visit every element exactly once.

Time Complexity

Operation Array List Singly List Doubly List Circular List
Insert Begin O(n) O(1) O(1) O(n)*
Insert Middle O(n) O(n) O(n) O(n)
Insert End O(1)** O(n) O(n) O(n)
Delete Begin O(n) O(1) O(1) O(n)*
Delete Middle O(n) O(n) O(n) O(n)
Delete End O(1) O(n) O(n) O(n)
Search O(n) O(n) O(n) O(n)
Traversal O(n) O(n) O(n) O(n)

* Can be O(1) if a suitable tail pointer is maintained.
** Amortized O(1) for a dynamic array.

7. Application – Polynomial Addition

Linked lists can represent polynomials efficiently. Each node stores: coefficient, exponent, next pointer.

Example: 5x³ + 4x² + 2 can be represented as [5,3] → [4,2] → [2,0].

Polynomial Representation

Polynomial 1:

5x³ + 4x² + 2

[5 | 3] → [4 | 2] → [2 | 0] → NULL


Polynomial 2:

3x³ + 2x + 7

[3 | 3] → [2 | 1] → [7 | 0] → NULL


Result:

8x³ + 4x² + 2x + 9

Pseudocode

POLYNOMIAL_ADD(P, Q)

    R ← EMPTY LIST

    WHILE P ≠ NULL AND Q ≠ NULL

        IF P.exponent = Q.exponent
            coefficient ← P.coefficient + Q.coefficient

            IF coefficient ≠ 0
                INSERT(R, coefficient, P.exponent)

            P ← P.next
            Q ← Q.next

        ELSE IF P.exponent > Q.exponent
            INSERT(R, P.coefficient, P.exponent)
            P ← P.next

        ELSE
            INSERT(R, Q.coefficient, Q.exponent)
            Q ← Q.next

    WHILE P ≠ NULL
        INSERT(R, P.coefficient, P.exponent)
        P ← P.next

    WHILE Q ≠ NULL
        INSERT(R, Q.coefficient, Q.exponent)
        Q ← Q.next

    RETURN R

C++ Implementation

#include <iostream>
using namespace std;

struct Term {
    int coeff;
    int exp;
    Term* next;

    Term(int c, int e) {
        coeff = c;
        exp = e;
        next = nullptr;
    }
};

void insertTerm(Term*& head, int coeff, int exp) {

    Term* newNode = new Term(coeff, exp);

    if (head == nullptr) {
        head = newNode;
        return;
    }

    Term* temp = head;

    while (temp->next != nullptr)
        temp = temp->next;

    temp->next = newNode;
}

Term* addPolynomial(Term* p, Term* q) {

    Term* result = nullptr;

    while (p != nullptr && q != nullptr) {

        if (p->exp == q->exp) {

            int sum = p->coeff + q->coeff;

            if (sum != 0)
                insertTerm(result, sum, p->exp);

            p = p->next;
            q = q->next;
        }

        else if (p->exp > q->exp) {

            insertTerm(result, p->coeff, p->exp);
            p = p->next;
        }

        else {

            insertTerm(result, q->coeff, q->exp);
            q = q->next;
        }
    }

    while (p != nullptr) {
        insertTerm(result, p->coeff, p->exp);
        p = p->next;
    }

    while (q != nullptr) {
        insertTerm(result, q->coeff, q->exp);
        q = q->next;
    }

    return result;
}

void display(Term* head) {

    while (head != nullptr) {

        cout << head->coeff
             << "x^"
             << head->exp;

        if (head->next != nullptr)
            cout << " + ";

        head = head->next;
    }

    cout << endl;
}

P2: 3x³ + 2x + 7
Result: 8x³ + 4x² + 2x + 9

8. Application – Browsing History

A linked list can be used to represent browser history. A doubly linked list is particularly suitable because the user can move both backward and forward.

Concept

[Google]
   ⇅
[YouTube]
   ⇅
[Wikipedia]
   ⇅
[College Portal]

        ↑
      CURRENT

Back:
CURRENT → previous node

Forward:
CURRENT → next node

Pseudocode

VISIT(url)

    newNode ← CREATE NODE(url)

    IF current.next ≠ NULL
        DELETE all forward history

    newNode.prev ← current
    current.next ← newNode
    current ← newNode

BACK()

    IF current.prev ≠ NULL
        current ← current.prev

FORWARD()

    IF current.next ≠ NULL
        current ← current.next

C++ Implementation

#include <iostream>
#include <string>
using namespace std;

struct HistoryNode {
    string url;
    HistoryNode* prev;
    HistoryNode* next;

    HistoryNode(string u) {
        url = u;
        prev = nullptr;
        next = nullptr;
    }
};

class BrowserHistory {
    HistoryNode* current;

public:

    BrowserHistory(string homepage) {
        current = new HistoryNode(homepage);
    }

    void visit(string url) {

        HistoryNode* newNode = new HistoryNode(url);

        newNode->prev = current;
        current->next = newNode;

        current = newNode;
    }

    void back() {

        if (current->prev != nullptr)
            current = current->prev;
    }

    void forward() {

        if (current->next != nullptr)
            current = current->next;
    }

    void showCurrent() {
        cout << "Current Page: "
             << current->url
             << endl;
    }
};

Why Doubly Linked List?

Back

Uses the previous pointer.

Forward

Uses the next pointer.

Dynamic

History can grow dynamically.

Easy Navigation

Both directions can be traversed.
