1. Graph ADT

A Graph is a non-linear data structure consisting of a set of vertices (nodes) and a set of edges connecting those vertices.

Vertex

A vertex represents an object or entity in a graph.

Edge

An edge represents a connection between two vertices.

Directed Graph

Edges have a direction: A → B.

Undirected Graph

Edges have no direction: A — B.

Weighted Graph

Each edge has a numerical weight or cost.

Connected Graph

Every vertex can be reached from another vertex.

Graph Representation

Representation Description Space
Adjacency Matrix 2D matrix representing edges O(V²)
Adjacency List List of neighbours for every vertex O(V + E)

C++ Graph ADT using Adjacency List

#include <iostream>
#include <vector>
using namespace std;

class Graph {
    int V;
    vector<vector<int>> adj;

public:
    Graph(int vertices) {
        V = vertices;
        adj.resize(V);
    }

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    void display() {
        for (int i = 0; i < V; i++) {
            cout << i << " : ";

            for (int v : adj[i])
                cout << v << " ";

            cout << endl;
        }
    }
};

int main() {
    Graph g(5);

    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 3);
    g.addEdge(2, 4);

    g.display();

    return 0;
}

2. Depth First Search (DFS)

DFS explores a graph by going as deep as possible before backtracking. It uses recursion or a stack.

DFS Steps

1. Start from a selected vertex.
2. Mark the vertex as visited.
3. Visit an unvisited adjacent vertex.
4. Continue recursively.
5. Backtrack when no unvisited neighbour exists.

DFS C++ Implementation

#include <iostream>
#include <vector>
using namespace std;

class Graph {
    int V;
    vector<vector<int>> adj;

    void DFSUtil(int v, vector<bool>& visited) {
        visited[v] = true;

        cout << v << " ";

        for (int u : adj[v]) {
            if (!visited[u])
                DFSUtil(u, visited);
        }
    }

public:

    Graph(int V) {
        this->V = V;
        adj.resize(V);
    }

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    void DFS(int start) {
        vector<bool> visited(V, false);
        DFSUtil(start, visited);
    }
};

int main() {

    Graph g(6);

    g.addEdge(0,1);
    g.addEdge(0,2);
    g.addEdge(1,3);
    g.addEdge(2,4);
    g.addEdge(3,5);
    g.addEdge(4,5);

    g.DFS(0);

    return 0;
}
Time: O(V + E)
Space: O(V)

3. Breadth First Search (BFS)

Breadth First Search (BFS) is a graph traversal algorithm that visits vertices level by level. BFS uses a Queue (FIFO) data structure.

Key idea: BFS first visits all immediate neighbours of a vertex, then visits the neighbours of those vertices.

BFS Traversal Example

Click "Run BFS" to start traversal.

How BFS Works

Step 1: Start at vertex 0.
Step 2: Put vertex 0 into the queue.
Step 3: Remove 0 and visit its neighbours 1 and 2.
Step 4: Process vertex 1 and visit 3.
Step 5: Process vertex 2 and visit 4.
Step 6: Process vertices 3 and 4 and visit 5.
Final BFS traversal: 0 → 1 → 2 → 3 → 4 → 5

BFS Data Structure

Operation Data Structure Purpose
Insert Queue Add newly discovered vertex
Remove Queue Process the next vertex
Visited Boolean Array Avoid visiting a vertex repeatedly

BFS C++ Implementation

#include <iostream>
#include <vector>
#include <queue>

using namespace std;

class Graph {

    int V;
    vector<vector<int>> adj;

public:

    Graph(int V) {

        this->V = V;
        adj.resize(V);
    }

    void addEdge(int u, int v) {

        adj[u].push_back(v);
        adj[v].push_back(u);
    }

    void BFS(int start) {

        vector<bool> visited(V, false);

        queue<int> q;

        // Start vertex
        visited[start] = true;
        q.push(start);

        cout << "BFS Traversal: ";

        while (!q.empty()) {

            int u = q.front();
            q.pop();

            cout << u << " ";

            // Visit all adjacent vertices
            for (int v : adj[u]) {

                if (!visited[v]) {

                    visited[v] = true;
                    q.push(v);
                }
            }
        }

        cout << endl;
    }
};

int main() {

    Graph g(6);

    g.addEdge(0, 1);
    g.addEdge(0, 2);
    g.addEdge(1, 3);
    g.addEdge(2, 4);
    g.addEdge(3, 5);
    g.addEdge(4, 5);

    g.BFS(0);

    return 0;
}
Time Complexity: O(V + E)
Space Complexity: O(V)

BFS vs DFS

Feature BFS DFS
Data Structure Queue Stack / Recursion
Traversal Level by level Depth first
Shortest path in unweighted graph Yes Not guaranteed
Time Complexity O(V + E) O(V + E)
Space Complexity O(V) O(V)
Typical Applications Shortest path, level order traversal, network broadcasting Cycle detection, connected components, topological sorting

BFS Applications

1. Shortest Path

BFS can find the shortest path in an unweighted graph.

2. Social Networks

Find people within a certain number of connections.

3. Web Crawlers

Explore web pages level by level.

4. Network Broadcasting

Send information to neighbouring nodes level by level.

5. Maze Solving

Find the shortest route through an unweighted maze.

6. Level Order Traversal

BFS is used for level-order traversal of trees.

5. Dijkstra's Shortest Path Algorithm

Dijkstra's algorithm finds the shortest distance from a source vertex to all other vertices in a weighted graph with non-negative edge weights.

Important: Dijkstra's algorithm should not be used when the graph contains negative edge weights.

6. Articulation Point

An Articulation Point, also called a cut vertex, is a vertex whose removal increases the number of connected components of an undirected graph.

7. Strongly Connected Components

A Strongly Connected Component (SCC) is a maximal set of vertices in a directed graph where every vertex is reachable from every other vertex.

Kosaraju's Algorithm

1. Perform DFS on the original graph.
2. Push vertices onto a stack according to finishing time.
3. Reverse every edge of the graph.
4. Pop vertices from the stack.
5. Run DFS on the reversed graph.
6. Every DFS traversal gives one SCC.

Example

Consider the directed graph: 

0 → 1
1 → 2
2 → 0

2 → 3

3 → 4
4 → 3

The SCCs are:

SCC 1

{0, 1, 2}

SCC 2

{3, 4}

C++ Implementation – Kosaraju's Algorithm

#include <iostream>
#include <vector>
#include <stack>
using namespace std;

class Graph {

    int V;

    vector<vector<int>> adj;

    void DFS1(int u,
              vector<bool>& visited,
              stack<int>& st) {

        visited[u] = true;

        for (int v : adj[u]) {

            if (!visited[v])
                DFS1(v, visited, st);
        }

        st.push(u);
    }

    void DFS2(int u,
              vector<bool>& visited,
              vector<vector<int>>& transpose) {

        visited[u] = true;

        cout << u << " ";

        for (int v : transpose[u]) {

            if (!visited[v])
                DFS2(v, visited, transpose);
        }
    }

public:

    Graph(int V) {

        this->V = V;
        adj.resize(V);
    }

    void addEdge(int u, int v) {

        adj[u].push_back(v);
    }

    void findSCC() {

        stack<int> st;

        vector<bool> visited(V, false);

        // First DFS
        for (int i = 0; i < V; i++) {

            if (!visited[i])
                DFS1(i, visited, st);
        }

        // Create transpose graph
        vector<vector<int>> transpose(V);

        for (int u = 0; u < V; u++) {

            for (int v : adj[u]) {

                transpose[v].push_back(u);
            }
        }

        // Second DFS
        fill(visited.begin(), visited.end(), false);

        cout << "Strongly Connected Components:\n";

        while (!st.empty()) {

            int u = st.top();
            st.pop();

            if (!visited[u]) {

                DFS2(u, visited, transpose);

                cout << endl;
            }
        }
    }
};

int main() {

    Graph g(5);

    g.addEdge(0,1);
    g.addEdge(1,2);
    g.addEdge(2,0);

    g.addEdge(2,3);

    g.addEdge(3,4);
    g.addEdge(4,3);

    g.findSCC();

    return 0;
}
Time: O(V + E)
Space: O(V + E)
