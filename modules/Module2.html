1. Stack and Queue ADTs

Stack

A stack is a linear data structure that follows LIFO – Last In, First Out.

Queue

A queue follows FIFO – First In, First Out.

Stack Applications

Function calls, undo operations, expression evaluation, parentheses matching and string reversal.

Queue Applications

CPU scheduling, printer queues, buffering, BFS and customer service systems.

Basic Stack Operations

PUSH(x) – Insert x at the top
POP() – Remove the top element
PEEK() – Read the top element
isEmpty() – Check whether stack is empty

Basic Queue Operations

ENQUEUE(x) – Insert x at rear
DEQUEUE() – Remove element from front
FRONT() – Read front element
isEmpty() – Check whether queue is empty

2. ArrayStack ADT

An ArrayStack uses an array to store elements. A variable top indicates the position of the top element.

TOP

Array Stack Pseudocode

PUSH(x)

IF top = MAX - 1
    PRINT "Stack Overflow"
    RETURN

top ← top + 1
stack[top] ← x


POP()

IF top = -1
    PRINT "Stack Underflow"
    RETURN

x ← stack[top]
top ← top - 1
RETURN x


PEEK()

IF top = -1
    PRINT "Stack Empty"
    RETURN

RETURN stack[top]

C++ Implementation

#include <iostream>
using namespace std;

class ArrayStack {

    int stack[100];
    int top;

public:

    ArrayStack() {
        top = -1;
    }

    void push(int x) {

        if (top == 99) {
            cout << "Stack Overflow";
            return;
        }

        stack[++top] = x;
    }

    void pop() {

        if (top == -1) {
            cout << "Stack Underflow";
            return;
        }

        cout << "Deleted: "
             << stack[top--];
    }

    int peek() {

        if (top == -1)
            return -1;

        return stack[top];
    }

    bool isEmpty() {
        return top == -1;
    }
};

3. LinkedListStack ADT

A linked stack uses linked nodes. The top of the stack is represented by the head pointer.

Representation

TOP
 ↓
[30|next] → [20|next] → [10|NULL]

Pseudocode

PUSH(x)

newNode ← CREATE NODE(x)

newNode.next ← top

top ← newNode


POP()

IF top = NULL
    PRINT "Stack Underflow"
    RETURN

temp ← top

x ← top.data

top ← top.next

DELETE temp

RETURN x


PEEK()

IF top = NULL
    RETURN EMPTY

RETURN top.data

C++ Implementation

#include <iostream>
using namespace std;

struct Node {

    int data;
    Node* next;

    Node(int x) {
        data = x;
        next = nullptr;
    }
};

class LinkedStack {

    Node* top;

public:

    LinkedStack() {
        top = nullptr;
    }

    void push(int x) {

        Node* newNode = new Node(x);

        newNode->next = top;

        top = newNode;
    }

    void pop() {

        if (top == nullptr) {
            cout << "Stack Underflow";
            return;
        }

        Node* temp = top;

        cout << "Deleted: "
             << top->data;

        top = top->next;

        delete temp;
    }

    int peek() {

        if (top == nullptr)
            return -1;

        return top->data;
    }
};

4. ArrayQueue ADT

An array queue stores elements using an array with front and rear.

Visualization

FRONT                         REAR
 ↓                              ↓
[10] → [20] → [30] → [40]
 ↑                              ↑
Remove                         Insert

Pseudocode

ENQUEUE(x)

IF rear = MAX - 1
    PRINT "Queue Overflow"
    RETURN

IF front = -1
    front ← 0

rear ← rear + 1
queue[rear] ← x


DEQUEUE()

IF front = -1 OR front > rear
    PRINT "Queue Underflow"
    RETURN

x ← queue[front]
front ← front + 1

RETURN x


FRONT()

IF front = -1
    RETURN EMPTY

RETURN queue[front]

C++ Implementation

#include <iostream>
using namespace std;

class ArrayQueue {

    int q[100];
    int front, rear;

public:

    ArrayQueue() {
        front = -1;
        rear = -1;
    }

    void enqueue(int x) {

        if (rear == 99) {
            cout << "Queue Overflow";
            return;
        }

        if (front == -1)
            front = 0;

        q[++rear] = x;
    }

    void dequeue() {

        if (front == -1 || front > rear) {
            cout << "Queue Underflow";
            return;
        }

        cout << "Deleted: "
             << q[front++];

        if (front > rear) {
            front = -1;
            rear = -1;
        }
    }

    int getFront() {

        if (front == -1)
            return -1;

        return q[front];
    }
};

5. LinkedQueue ADT

A linked queue uses nodes and maintains two pointers: front and rear.

Representation

FRONT                                REAR
 ↓                                      ↓
[10|next] → [20|next] → [30|NULL]

Pseudocode

ENQUEUE(x)

newNode ← CREATE NODE(x)

IF rear = NULL
    front ← rear ← newNode
    RETURN

rear.next ← newNode
rear ← newNode


DEQUEUE()

IF front = NULL
    PRINT "Queue Underflow"
    RETURN

temp ← front

front ← front.next

IF front = NULL
    rear ← NULL

DELETE temp

C++ Implementation

#include <iostream>
using namespace std;

struct QNode {

    int data;
    QNode* next;

    QNode(int x) {
        data = x;
        next = nullptr;
    }
};

class LinkedQueue {

    QNode* front;
    QNode* rear;

public:

    LinkedQueue() {
        front = rear = nullptr;
    }

    void enqueue(int x) {

        QNode* newNode = new QNode(x);

        if (rear == nullptr) {
            front = rear = newNode;
            return;
        }

        rear->next = newNode;
        rear = newNode;
    }

    void dequeue() {

        if (front == nullptr) {
            cout << "Queue Underflow";
            return;
        }

        QNode* temp = front;

        cout << "Deleted: "
             << front->data;

        front = front->next;

        if (front == nullptr)
            rear = nullptr;

        delete temp;
    }

    int getFront() {

        if (front == nullptr)
            return -1;

        return front->data;
    }
};

6. Application – String Reverse Using Stack

A stack is ideal for reversing a string because it follows LIFO.

Example:

Input: DATA

Push: D → A → T → A

Pop: A → T → A → D

Output: ATAD

Pseudocode

REVERSE_STRING(str)

Create empty stack S

FOR each character c in str
    PUSH(S, c)

result ← ""

WHILE S is not empty
    result ← result + POP(S)

RETURN result

C++ Implementation

#include <iostream>
#include <stack>
#include <string>
using namespace std;

string reverseString(string str) {

    stack<char> s;

    for (char c : str)
        s.push(c);

    string result = "";

    while (!s.empty()) {

        result += s.top();

        s.pop();
    }

    return result;
}

7. Application – Expression Evaluation

Stacks are widely used to evaluate arithmetic expressions. Operands are stored in an operand stack while operators are processed according to precedence.

Operator Precedence

Operator Precedence
() Highest
^ 3
* / % 2
+ - 1

Infix → Postfix

Infix:

A + B * C

Step 1:
Read A
Output A

Step 2:
Read +
Push +

Step 3:
Read B
Output B

Step 4:
Read *
Push *

Step 5:
Read C
Output C

Pop remaining operators

Postfix:

A B C * +

Pseudocode – Infix to Postfix

INFIX_TO_POSTFIX(expression)

Create empty operator stack

FOR each token x

    IF x is operand
        Add x to output

    ELSE IF x = '('
        PUSH x

    ELSE IF x = ')'
        WHILE top ≠ '('
            output ← POP
        POP '('

    ELSE
        WHILE stack not empty AND
              precedence(top) ≥ precedence(x)

            output ← POP

        PUSH x

WHILE stack not empty
    output ← POP

RETURN output

C++ Implementation

#include <iostream>
#include <stack>
#include <string>
#include <cctype>

using namespace std;

int precedence(char op) {

    if (op == '^')
        return 3;

    if (op == '*' || op == '/' || op == '%')
        return 2;

    if (op == '+' || op == '-')
        return 1;

    return 0;
}

string infixToPostfix(string exp) {

    stack<char> s;
    string result;

    for (char c : exp) {

        if (isalnum(c)) {

            result += c;
        }

        else if (c == '(') {

            s.push(c);
        }

        else if (c == ')') {

            while (!s.empty() && s.top() != '(') {

                result += s.top();
                s.pop();
            }

            if (!s.empty())
                s.pop();
        }

        else {

            while (!s.empty() &&
                   precedence(s.top()) >= precedence(c)) {

                result += s.top();
                s.pop();
            }

            s.push(c);
        }
    }

    while (!s.empty()) {

        result += s.top();
        s.pop();
    }

    return result;
}

Postfix Evaluation

Evaluate:

2 3 * 4 +

Step 1:
Push 2

Stack:
2

Step 2:
Push 3

Stack:
2 3

Step 3:
*
Pop 3 and 2
2 × 3 = 6
Push 6

Stack:
6

Step 4:
Push 4

Stack:
6 4

Step 5:
+
6 + 4 = 10

Answer = 10

C++ – Postfix Evaluation

#include <iostream>
#include <stack>
#include <cctype>
using namespace std;

int evaluatePostfix(string exp) {

    stack<int> s;

    for (char c : exp) {

        if (isdigit(c)) {

            s.push(c - '0');
        }

        else {

            int b = s.top();
            s.pop();

            int a = s.top();
            s.pop();

            switch(c) {

                case '+':
                    s.push(a + b);
                    break;

                case '-':
                    s.push(a - b);
                    break;

                case '*':
                    s.push(a * b);
                    break;

                case '/':
                    s.push(a / b);
                    break;
            }
        }
    }

    return s.top();
}

8. Circular Queue ADT

A circular queue connects the last array position back to the first position. This allows previously unused spaces to be reused.

Circular movement:

rear = (rear + 1) % MAX
front = (front + 1) % MAX

Pseudocode

ENQUEUE(x)

IF (rear + 1) MOD MAX = front
    PRINT "Queue Full"
    RETURN

IF front = -1
    front ← rear ← 0

ELSE
    rear ← (rear + 1) MOD MAX

queue[rear] ← x


DEQUEUE()

IF front = -1
    PRINT "Queue Empty"
    RETURN

x ← queue[front]

IF front = rear
    front ← rear ← -1

ELSE
    front ← (front + 1) MOD MAX

RETURN x

C++ Implementation

#include <iostream>
using namespace std;

class CircularQueue {

    int q[5];

    int front;
    int rear;

public:

    CircularQueue() {

        front = -1;
        rear = -1;
    }

    bool isFull() {

        return (rear + 1) % 5 == front;
    }

    bool isEmpty() {

        return front == -1;
    }

    void enqueue(int x) {

        if (isFull()) {

            cout << "Queue Full";
            return;
        }

        if (front == -1)
            front = rear = 0;

        else
            rear = (rear + 1) % 5;

        q[rear] = x;
    }

    void dequeue() {

        if (isEmpty()) {

            cout << "Queue Empty";
            return;
        }

        cout << q[front];

        if (front == rear)

            front = rear = -1;

        else

            front = (front + 1) % 5;
    }
};

9. Priority Queue ADT

A priority queue is a queue in which each element is associated with a priority. The element with the highest priority is served first.

Example:

Patient A – Priority 2
Patient B – Priority 5
Patient C – Priority 1
Patient D – Priority 4

Removal order: B → D → A → C

Interactive Priority Queue

Pseudocode

INSERT(item, priority)

Create new element

Add element to priority queue

Arrange elements according to priority


DELETE()

IF queue is empty
    PRINT "Queue Empty"
    RETURN

Find element with highest priority

Remove it

RETURN element


PEEK()

Find highest-priority element

Return it

C++ Implementation – Array Based Priority Queue

#include <iostream>
using namespace std;

struct Item {

    int data;
    int priority;
};

class PriorityQueue {

    Item pq[100];
    int n;

public:

    PriorityQueue() {
        n = 0;
    }

    void insert(int data, int priority) {

        pq[n].data = data;
        pq[n].priority = priority;

        n++;
    }

    int deleteHighest() {

        if (n == 0)
            return -1;

        int index = 0;

        for (int i = 1; i < n; i++) {

            if (pq[i].priority >
                pq[index].priority)

                index = i;
        }

        int value = pq[index].data;

        for (int i = index; i < n - 1; i++)

            pq[i] = pq[i + 1];

        n--;

        return value;
    }

    int peek() {

        if (n == 0)
            return -1;

        int index = 0;

        for (int i = 1; i < n; i++) {

            if (pq[i].priority >
                pq[index].priority)

                index = i;
        }

        return pq[index].data;
    }
};

Applications of Priority Queue

CPU Scheduling

High-priority processes can execute first.

Hospital Emergency

Critical patients can receive priority.

Network Routing

Important packets can be processed first.

Operating Systems

Processes and tasks can be scheduled according to priority.

10. Stack and Queue Comparison

Structure Rule Insertion Deletion Typical Application
Array Stack LIFO Top Top Expression evaluation
Linked Stack LIFO Top Top Undo operations
Array Queue FIFO Rear Front Simple scheduling
Linked Queue FIFO Rear Front Dynamic queues
Circular Queue FIFO Rear Front Buffer management
Priority Queue Priority According to priority Highest priority CPU scheduling

Time Complexity

Operation Array Stack Linked Stack Array Queue Linked Queue Circular Queue
Insert O(1) O(1) O(1) O(1) O(1)
Delete O(1) O(1) O(1) O(1) O(1)
Peek O(1) O(1) O(1) O(1) O(1)

Key Differences

Stack

LIFO

Last inserted element is removed first.

Queue

FIFO

First inserted element is removed first.

Circular Queue

Reuses empty positions by wrapping around.

Priority Queue

Element with highest priority is processed first.
